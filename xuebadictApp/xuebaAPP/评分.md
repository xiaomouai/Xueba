```python
"""
音节数据模型

表示一个单词拆解后的音节结构，
包含每个音节的文本、音标、重音位置等信息。
"""

from dataclasses import dataclass, field
from typing import Optional


@dataclass
class Syllable:
    """单个音节"""
    text: str               # 音节文本，如 "ap"
    phoneme: str             # 音标，如 "æp"
    stress: int = 0          # 重音级别：0=无重音, 1=主重音, 2=次重音
    index: int = 0           # 音节在单词中的位置索引

    def to_dict(self) -> dict:
        return {
            "text": self.text,
            "phoneme": self.phoneme,
            "stress": self.stress,
            "index": self.index,
        }


@dataclass
class WordSyllableInfo:
    """单词的完整音节信息"""
    word: str                          # 原始单词
    phonetic: str                      # 完整音标，如 "/ˈæp.əl/"
    phonetic_us: Optional[str] = None  # 美式音标
    phonetic_uk: Optional[str] = None  # 英式音标
    syllables: list[Syllable] = field(default_factory=list)
    syllable_count: int = 0            # 音节数量
    stress_pattern: str = ""           # 重音模式，如 "10"（第一个重读）
    audio_url: Optional[str] = None    # 标准读音URL

    def to_dict(self) -> dict:
        return {
            "word": self.word,
            "phonetic": self.phonetic,
            "phonetic_us": self.phonetic_us,
            "phonetic_uk": self.phonetic_uk,
            "syllables": [s.to_dict() for s in self.syllables],
            "syllable_count": self.syllable_count,
            "stress_pattern": self.stress_pattern,
            "audio_url": self.audio_url,
        }
```

```python
"""
发音评测结果模型
"""

from dataclasses import dataclass, field
from typing import Optional
from enum import Enum


class ScoreLevel(str, Enum):
    """评分等级"""
    EXCELLENT = "excellent"    # 90-100
    GOOD = "good"              # 70-89
    FAIR = "fair"              # 50-69
    POOR = "poor"              # 0-49


@dataclass
class SyllableScore:
    """单个音节的评分"""
    syllable_index: int
    syllable_text: str
    expected_phoneme: str      # 期望音标
    recognized_phoneme: str    # 识别到的音标
    score: float               # 0-100
    is_correct: bool
    feedback: str = ""         # 针对该音节的反馈

    def to_dict(self) -> dict:
        return {
            "syllable_index": self.syllable_index,
            "syllable_text": self.syllable_text,
            "expected_phoneme": self.expected_phoneme,
            "recognized_phoneme": self.recognized_phoneme,
            "score": round(self.score, 1),
            "is_correct": self.is_correct,
            "feedback": self.feedback,
        }


@dataclass
class PronunciationResult:
    """完整发音评测结果"""
    word: str
    recognized_text: str            # 语音识别出的文本
    overall_score: float            # 总分 0-100
    level: ScoreLevel               # 评分等级
    accuracy_score: float           # 准确度分
    fluency_score: float            # 流利度分
    syllable_scores: list[SyllableScore] = field(default_factory=list)
    feedback: str = ""              # 总体反馈
    suggestions: list[str] = field(default_factory=list)  # 改进建议
    is_correct: bool = False        # 是否完全正确

    def to_dict(self) -> dict:
        return {
            "word": self.word,
            "recognized_text": self.recognized_text,
            "overall_score": round(self.overall_score, 1),
            "level": self.level.value,
            "accuracy_score": round(self.accuracy_score, 1),
            "fluency_score": round(self.fluency_score, 1),
            "syllable_scores": [s.to_dict() for s in self.syllable_scores],
            "feedback": self.feedback,
            "suggestions": self.suggestions,
            "is_correct": self.is_correct,
        }

    @staticmethod
    def get_level(score: float) -> ScoreLevel:
        if score >= 90:
            return ScoreLevel.EXCELLENT
        elif score >= 70:
            return ScoreLevel.GOOD
        elif score >= 50:
            return ScoreLevel.FAIR
        else:
            return ScoreLevel.POOR
```

### 4. 音节拆解服务 `services/syllable_service.py`

```python
"""
音节拆解服务

功能：
- 将英文单词拆解为音节
- 提供每个音节的音标和重音标记
- 使用 CMU Pronouncing Dictionary + 规则引擎

依赖：
  pip install nltk pronouncing eng-to-ipa
"""

import re
import logging
from typing import Optional

import pronouncing  # 基于CMU词典
import eng_to_ipa   # 英文转IPA音标

from models.syllable import Syllable, WordSyllableInfo

logger = logging.getLogger(__name__)


# ==================== 音节拆解规则引擎 ====================

# 元音字母组合（用于规则兜底）
_VOWEL_PATTERNS = re.compile(
    r'[aeiou]+|'           # 纯元音
    r'[aeiou][rl]|'        # 元音+r/l
    r'y(?=[^aeiou]|$)',    # y在辅音前或词尾作元音
    re.IGNORECASE
)

# 常见不拆分的辅音组合
_CONSONANT_CLUSTERS = {
    "bl", "br", "cl", "cr", "dr", "fl", "fr", "gl", "gr",
    "pl", "pr", "sc", "sk", "sl", "sm", "sn", "sp", "st",
    "str", "sw", "tr", "tw", "spr", "spl", "scr",
    "ch", "sh", "th", "wh", "ph", "ck", "ng", "nk",
}


def _get_cmu_phones(word: str) -> Optional[list[str]]:
    """
    从CMU词典获取音素列表

    CMU格式示例：
    "apple" -> ['AE1', 'P', 'AH0', 'L']
    数字表示重音：0=无, 1=主重音, 2=次重音
    """
    phones_list = pronouncing.phones_for_word(word.lower())
    if phones_list:
        return phones_list[0].split()
    return None


def _phones_to_syllables(word: str, phones: list[str]) -> list[Syllable]:
    """
    将CMU音素序列拆分为音节

    音节拆分规则：
    1. 每个元音音素（带数字后缀）是一个音节核心
    2. 两个元音之间的辅音，归后一个音节（最大起始原则）
    3. 词首辅音归第一个音节
    """
    # 标记元音位置
    vowel_indices = []
    for i, phone in enumerate(phones):
        if phone[-1].isdigit():  # CMU中元音末尾有数字
            vowel_indices.append(i)

    if not vowel_indices:
        # 无元音（极少见），整个词作为一个音节
        return [Syllable(text=word, phoneme="".join(phones), stress=0, index=0)]

    # 按元音分组音素
    syllable_phones: list[list[str]] = []
    prev_end = 0
    for idx, vowel_idx in enumerate(vowel_indices):
        if idx == len(vowel_indices) - 1:
            # 最后一个元音，取到末尾
            syllable_phones.append(phones[prev_end:])
        else:
            # 找下一个元音之前的辅音分界点
            next_vowel = vowel_indices[idx + 1]
            # 辅音群在两个元音之间，用最大起始原则
            consonants_between = phones[vowel_idx + 1:next_vowel]
            split_point = vowel_idx + 1
            
            if len(consonants_between) > 1:
                # 把尽可能多的辅音留给下一个音节
                split_point = next_vowel - len(consonants_between) + 1
            elif len(consonants_between) == 1:
                split_point = vowel_idx + 1  # 单辅音归后
            
            syllable_phones.append(phones[prev_end:split_point])
            prev_end = split_point

    # 将字母分配给音节（近似映射）
    syllable_texts = _distribute_letters(word, len(syllable_phones))

    # 构建音节对象
    syllables = []
    for i, (syl_phones, syl_text) in enumerate(zip(syllable_phones, syllable_texts)):
        # 提取重音
        stress = 0
        for phone in syl_phones:
            if phone[-1] == "1":
                stress = 1
            elif phone[-1] == "2" and stress == 0:
                stress = 2

        # 转换音素为IPA（简化）
        phoneme = _cmu_to_ipa_simple(syl_phones)

        syllables.append(Syllable(
            text=syl_text,
            phoneme=phoneme,
            stress=stress,
            index=i,
        ))

    return syllables


def _distribute_letters(word: str, num_syllables: int) -> list[str]:
    """
    将单词字母近似分配到各音节

    使用元音检测进行分割，如果音节数不匹配则均匀分配
    """
    if num_syllables <= 1:
        return [word]

    # 找到元音位置
    vowel_positions = [i for i, c in enumerate(word.lower()) if c in "aeiouy"]

    if len(vowel_positions) < num_syllables:
        # 均匀分割
        chunk_size = max(1, len(word) // num_syllables)
        parts = []
        for i in range(num_syllables):
            start = i * chunk_size
            end = start + chunk_size if i < num_syllables - 1 else len(word)
            parts.append(word[start:end])
        return parts

    # 基于元音分割
    parts = []
    prev = 0
    for i in range(num_syllables - 1):
        if i < len(vowel_positions) - 1:
            # 在两个元音之间的辅音处分割
            curr_vowel = vowel_positions[i]
            next_vowel = vowel_positions[i + 1]
            # 分割点在两个元音中间的辅音位置
            split = (curr_vowel + next_vowel + 1) // 2
            parts.append(word[prev:split])
            prev = split
    parts.append(word[prev:])

    return parts


# CMU音素到IPA的简化映射
_CMU_TO_IPA = {
    "AA": "ɑː", "AE": "æ", "AH": "ʌ", "AO": "ɔː", "AW": "aʊ",
    "AY": "aɪ", "EH": "ɛ", "ER": "ɜːr", "EY": "eɪ", "IH": "ɪ",
    "IY": "iː", "OW": "oʊ", "OY": "ɔɪ", "UH": "ʊ", "UW": "uː",
    "B": "b", "CH": "tʃ", "D": "d", "DH": "ð", "F": "f",
    "G": "ɡ", "HH": "h", "JH": "dʒ", "K": "k", "L": "l",
    "M": "m", "N": "n", "NG": "ŋ", "P": "p", "R": "r",
    "S": "s", "SH": "ʃ", "T": "t", "TH": "θ", "V": "v",
    "W": "w", "Y": "j", "Z": "z", "ZH": "ʒ",
}


def _cmu_to_ipa_simple(phones: list[str]) -> str:
    """将CMU音素列表转换为IPA字符串"""
    ipa_parts = []
    for phone in phones:
        # 去掉重音数字
        base = phone.rstrip("012")
        ipa = _CMU_TO_IPA.get(base, base.lower())
        ipa_parts.append(ipa)
    return "".join(ipa_parts)


def _get_ipa_phonetic(word: str) -> str:
    """获取单词的完整IPA音标"""
    try:
        ipa = eng_to_ipa.convert(word)
        if ipa and "*" not in ipa:  # eng_to_ipa 未找到时返回带*的原文
            return f"/{ipa}/"
    except Exception:
        pass

    # 降级：从CMU转换
    phones = _get_cmu_phones(word)
    if phones:
        return f"/{_cmu_to_ipa_simple(phones)}/"
    return ""


# ==================== 公开接口 ====================

def analyze_word(word: str) -> WordSyllableInfo:
    """
    分析单词的音节结构

    Args:
        word: 英文单词

    Returns:
        WordSyllableInfo 包含完整的音节拆解信息
    """
    word_clean = word.strip().lower()
    logger.info(f"[SyllableService] 分析单词: '{word_clean}'")

    # 获取CMU音素
    phones = _get_cmu_phones(word_clean)

    if phones:
        syllables = _phones_to_syllables(word_clean, phones)
    else:
        # CMU词典中没有，使用音节计数估算
        logger.warning(f"[SyllableService] CMU词典未收录: '{word_clean}', 使用规则引擎")
        count = pronouncing.syllable_count(
            pronouncing.phones_for_word(word_clean)[0]
        ) if pronouncing.phones_for_word(word_clean) else _estimate_syllable_count(word_clean)
        
        texts = _distribute_letters(word_clean, max(1, count))
        syllables = [
            Syllable(text=t, phoneme="", stress=0, index=i)
            for i, t in enumerate(texts)
        ]

    # 获取IPA音标
    phonetic = _get_ipa_phonetic(word_clean)

    # 构建重音模式
    stress_pattern = "".join(str(s.stress) for s in syllables)

    result = WordSyllableInfo(
        word=word_clean,
        phonetic=phonetic,
        syllables=syllables,
        syllable_count=len(syllables),
        stress_pattern=stress_pattern,
    )

    logger.info(
        f"[SyllableService] 结果: {word_clean} → "
        f"{[s.text for s in syllables]} ({phonetic})"
    )
    return result


def _estimate_syllable_count(word: str) -> int:
    """基于规则估算音节数"""
    word = word.lower()
    count = len(re.findall(r'[aeiouy]+', word))
    # 词尾的 silent e
    if word.endswith("e") and count > 1:
        count -= 1
    # 词尾 -le 算一个音节
    if word.endswith("le") and len(word) > 2 and word[-3] not in "aeiouy":
        count += 1
    return max(1, count)
```

### 5. 发音评测服务 `services/pronunciation_service.py`

```python
"""
发音评测业务编排服务

职责：
- 接收音频数据
- 调用语音识别引擎
- 与预期发音对比
- 生成评分和反馈
"""

import os
import base64
import tempfile
import logging
from typing import Optional

from models.evaluation import PronunciationResult, SyllableScore, ScoreLevel
from models.syllable import WordSyllableInfo
from services.syllable_service import analyze_word
from engines import get_evaluator_engine

logger = logging.getLogger(__name__)


def evaluate_pronunciation(
    word: str,
    audio_base64: str,
    audio_format: str = "wav",
) -> PronunciationResult:
    """
    评测发音

    Args:
        word: 目标单词
        audio_base64: Base64编码的音频数据
        audio_format: 音频格式

    Returns:
        PronunciationResult 评测结果
    """
    word = word.strip().lower()
    logger.info(f"[PronunciationService] 评测: '{word}'")

    # 1. 解码并保存音频
    temp_path = _save_audio_temp(audio_base64, audio_format)

    try:
        # 2. 语音识别
        engine = get_evaluator_engine()
        recognized_text = engine.recognize(temp_path)
        logger.info(f"[PronunciationService] 识别结果: '{recognized_text}'")

        # 3. 获取音节信息
        syllable_info = analyze_word(word)

        # 4. 计算评分
        result = _compute_scores(word, recognized_text, syllable_info)

        return result

    finally:
        # 清理临时文件
        if os.path.exists(temp_path):
            os.unlink(temp_path)


def _save_audio_temp(audio_base64: str, fmt: str) -> str:
    """Base64音频解码并保存为临时文件"""
    audio_data = base64.b64decode(audio_base64)
    with tempfile.NamedTemporaryFile(
        suffix=f".{fmt}", delete=False, dir=None
    ) as f:
        f.write(audio_data)
        return f.name


def _compute_scores(
    word: str,
    recognized: str,
    syllable_info: WordSyllableInfo,
) -> PronunciationResult:
    """
    计算综合评分

    评分维度：
    1. 准确度：识别文本与目标单词的匹配度
    2. 流利度：基于语音特征（简化实现）
    3. 音节级评分：逐音节对比
    """
    recognized_clean = recognized.strip().lower()

    # ---- 准确度评分 ----
    if recognized_clean == word:
        accuracy_score = 100.0
    elif word in recognized_clean or recognized_clean in word:
        # 部分匹配
        accuracy_score = 70.0 + 30.0 * (
            len(set(word) & set(recognized_clean)) / max(len(set(word)), 1)
        )
    else:
        # 编辑距离评分
        accuracy_score = max(0, (1 - _edit_distance(word, recognized_clean) / max(len(word), 1)) * 100)

    # ---- 流利度评分（简化：基于识别置信度） ----
    fluency_score = min(100, accuracy_score + 10) if accuracy_score > 50 else accuracy_score * 0.8

    # ---- 音节级评分 ----
    syllable_scores = _score_syllables(syllable_info, recognized_clean)

    # ---- 综合评分 ----
    overall_score = accuracy_score * 0.6 + fluency_score * 0.2
    if syllable_scores:
        avg_syl = sum(s.score for s in syllable_scores) / len(syllable_scores)
        overall_score += avg_syl * 0.2
    else:
        overall_score += accuracy_score * 0.2

    overall_score = min(100, max(0, overall_score))
    level = PronunciationResult.get_level(overall_score)

    # ---- 生成反馈 ----
    feedback, suggestions = _generate_feedback(
        word, recognized_clean, overall_score, level, syllable_scores, syllable_info
    )

    return PronunciationResult(
        word=word,
        recognized_text=recognized_clean,
        overall_score=overall_score,
        level=level,
        accuracy_score=accuracy_score,
        fluency_score=fluency_score,
        syllable_scores=syllable_scores,
        feedback=feedback,
        suggestions=suggestions,
        is_correct=(accuracy_score >= 90),
    )


def _score_syllables(
    syllable_info: WordSyllableInfo,
    recognized: str,
) -> list[SyllableScore]:
    """逐音节评分"""
    scores = []
    # 简化实现：基于字符匹配
    rec_pos = 0

    for syl in syllable_info.syllables:
        syl_len = len(syl.text)
        rec_chunk = recognized[rec_pos:rec_pos + syl_len] if rec_pos < len(recognized) else ""
        rec_pos += syl_len

        if rec_chunk.lower() == syl.text.lower():
            score = 100.0
            is_correct = True
            feedback = "✅ 发音正确"
        elif any(c in rec_chunk.lower() for c in syl.text.lower()):
            score = 60.0
            is_correct = False
            feedback = f"💡 音节 '{syl.text}' 的发音需要注意"
        else:
            score = 20.0
            is_correct = False
            feedback = f"❌ 音节 '{syl.text}' /{syl.phoneme}/ 需要练习"

        scores.append(SyllableScore(
            syllable_index=syl.index,
            syllable_text=syl.text,
            expected_phoneme=syl.phoneme,
            recognized_phoneme=rec_chunk,
            score=score,
            is_correct=is_correct,
            feedback=feedback,
        ))

    return scores


def _generate_feedback(
    word: str,
    recognized: str,
    score: float,
    level: ScoreLevel,
    syllable_scores: list[SyllableScore],
    syllable_info: WordSyllableInfo,
) -> tuple[str, list[str]]:
    """生成反馈文本和改进建议"""
    
    feedback_map = {
        ScoreLevel.EXCELLENT: f"🎉 太棒了！'{word}' 发音非常标准！",
        ScoreLevel.GOOD: f"👍 不错！'{word}' 发音基本正确，还有小的提升空间。",
        ScoreLevel.FAIR: f"💪 继续加油！'{word}' 的发音还需要多练习。",
        ScoreLevel.POOR: f"📚 让我们一起来练习 '{word}' 的发音吧！",
    }
    feedback = feedback_map[level]

    suggestions = []

    # 基于音节分析的建议
    wrong_syllables = [s for s in syllable_scores if not s.is_correct]
    if wrong_syllables:
        for ws in wrong_syllables:
            suggestions.append(
                f"音节 '{ws.syllable_text}' 的发音是 /{ws.expected_phoneme}/，"
                f"注意口型和舌位"
            )

    # 重音建议
    stress_syllables = [s for s in syllable_info.syllables if s.stress == 1]
    if stress_syllables:
        stress_syl = stress_syllables[0]
        suggestions.append(
            f"注意重音在第{stress_syl.index + 1}个音节 '{stress_syl.text}' 上"
        )

    if not suggestions:
        suggestions.append("继续保持，多多练习！")

    return feedback, suggestions


def _edit_distance(s1: str, s2: str) -> int:
    """Levenshtein编辑距离"""
    m, n = len(s1), len(s2)
    dp = [[0] * (n + 1) for _ in range(m + 1)]
    for i in range(m + 1):
        dp[i][0] = i
    for j in range(n + 1):
        dp[0][j] = j
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            cost = 0 if s1[i-1] == s2[j-1] else 1
            dp[i][j] = min(dp[i-1][j] + 1, dp[i][j-1] + 1, dp[i-1][j-1] + cost)
    return dp[m][n]
```

```python
class GoogleEvaluatorEngine(BaseEvaluatorEngine):
    """使用 Google Web Speech API 进行语音识别"""

        self._recognizer = sr.Recognizer()

    def recognize(self, audio_path: str) -> str:

            return ""
```

```dart
import 'dart:convert';
import 'package:http/http.dart' as http;
import '../models/syllable_info.dart';
import '../models/pronunciation_result.dart';

/// 发音评测API服务
class PronunciationApiService {
  static const String _baseUrl = 'http://your-server.com/api/pronunciation';

  /// 获取单词音节拆解
  static Future<WordSyllableInfo> getSyllables(String word) async {
    final response = await http.get(
      Uri.parse('$_baseUrl/syllables/$word'),
    );

    if (response.statusCode == 200) {
      final data = jsonDecode(response.body)['data'];
      return WordSyllableInfo.fromJson(data);
    }
    throw Exception('获取音节信息失败: ${response.statusCode}');
  }

  /// 批量获取音节拆解
  static Future<List<WordSyllableInfo>> getBatchSyllables(
    List<String> words,
  ) async {
    final response = await http.post(
      Uri.parse('$_baseUrl/batch-syllables'),
      headers: {'Content-Type': 'application/json'},
      body: jsonEncode({'words': words}),
    );

    if (response.statusCode == 200) {
      final results = jsonDecode(response.body)['data']['results'] as List;
      return results
          .where((r) => r['error'] == null)
          .map((r) => WordSyllableInfo.fromJson(r))
          .toList();
    }
    throw Exception('批量获取失败: ${response.statusCode}');
  }

  /// 提交发音评测
  static Future<PronunciationResult> evaluate({
    required String word,
    required String audioBase64,
    String audioFormat = 'wav',
  }) async {
    final response = await http.post(
      Uri.parse('$_baseUrl/evaluate'),
      headers: {'Content-Type': 'application/json'},
      body: jsonEncode({
        'word': word,
        'audio_base64': audioBase64,
        'audio_format': audioFormat,
      }),
    );

    if (response.statusCode == 200) {
      final data = jsonDecode(response.body)['data'];
      return PronunciationResult.fromJson(data);
    }
    throw Exception('评测失败: ${response.statusCode}');
  }
}
```

```dart
import 'dart:convert';
import 'dart:io';

import 'package:flutter/material.dart';
import '../models/syllable_info.dart';
import '../models/pronunciation_result.dart';
import '../services/syllable_api_service.dart';

/// 录音状态
enum RecordingState { idle, recording, processing, evaluated }

/// 学习状态
enum LearningStep { viewWord, viewSyllable, practice, result }

/// 单词速记+发音评测状态管理
class WordMemoryProvider extends ChangeNotifier {
  // ==================== 单词列表 ====================
  final List<WordItem> _words;
  int _currentIndex = 0;

  // ==================== 音节信息 ====================
  WordSyllableInfo? _syllableInfo;
  bool _syllableLoading = false;

  // ==================== 录音评测 ====================
  RecordingState _recordingState = RecordingState.idle;
  PronunciationResult? _pronunciationResult;

  // ==================== 学习步骤 ====================
  LearningStep _learningStep = LearningStep.viewWord;

  // ==================== 构造 ====================
  WordMemoryProvider({required List<WordItem> words}) : _words = words;

  // ==================== Getters ====================
  WordItem get currentWord => _words[_currentIndex];
  int get currentIndex => _currentIndex;
  int get totalWords => _words.length;
  double get progress => (_currentIndex + 1) / _words.length;

  WordSyllableInfo? get syllableInfo => _syllableInfo;
  bool get syllableLoading => _syllableLoading;

  RecordingState get recordingState => _recordingState;
  PronunciationResult? get pronunciationResult => _pronunciationResult;

  LearningStep get learningStep => _learningStep;

  bool get hasNext => _currentIndex < _words.length - 1;
  bool get hasPrevious => _currentIndex > 0;

  // ==================== 导航 ====================

  /// 下一个单词
  void nextWord() {
    if (!hasNext) return;
    _currentIndex++;
    _resetWordState();
    notifyListeners();
  }

  /// 上一个单词
  void previousWord() {
    if (!hasPrevious) return;
    _currentIndex--;
    _resetWordState();
    notifyListeners();
  }

  /// 跳转到指定单词
  void goToWord(int index) {
    if (index < 0 || index >= _words.length) return;
    _currentIndex = index;
    _resetWordState();
    notifyListeners();
  }

  void _resetWordState() {
    _syllableInfo = null;
    _pronunciationResult = null;
    _recordingState = RecordingState.idle;
    _learningStep = LearningStep.viewWord;
  }

  // ==================== 学习步骤 ====================

  /// 进入音节学习
  Future<void> showSyllableBreakdown() async {
    _learningStep = LearningStep.viewSyllable;
    notifyListeners();

    if (_syllableInfo != null) return; // 已加载

    _syllableLoading = true;
    notifyListeners();

    try {
      _syllableInfo = await PronunciationApiService.getSyllables(
        currentWord.word,
      );
    } catch (e) {
      debugPrint('音节加载失败: $e');
    } finally {
      _syllableLoading = false;
      notifyListeners();
    }
  }

  /// 进入发音练习
  void enterPractice() {
    _learningStep = LearningStep.practice;
    _pronunciationResult = null;
    _recordingState = RecordingState.idle;
    notifyListeners();
  }

  // ==================== 录音评测 ====================

  /// 开始录音
  Future<void> startRecording() async {
    final success = await AudioRecorderService.start();
    if (success) {
      _recordingState = RecordingState.recording;
      notifyListeners();
    }
  }

  /// 停止录音并评测
  Future<void> stopAndEvaluate() async {
    if (_recordingState != RecordingState.recording) return;

    _recordingState = RecordingState.processing;
    notifyListeners();

    try {
      final audioPath = await AudioRecorderService.stop();
      if (audioPath == null) throw Exception('录音文件为空');

      // 读取音频文件并转Base64
      final bytes = await File(audioPath).readAsBytes();
      final base64Audio = base64Encode(bytes);

      // 调用评测API
      _pronunciationResult = await PronunciationApiService.evaluate(
        word: currentWord.word,
        audioBase64: base64Audio,
        audioFormat: 'wav',
      );

      _recordingState = RecordingState.evaluated;
      _learningStep = LearningStep.result;

      // 更新单词掌握状态
      if (_pronunciationResult!.isCorrect) {
        currentWord.markAsPracticed(true);
      }
    } catch (e) {
      _recordingState = RecordingState.idle;
      debugPrint('评测失败: $e');
    }

    notifyListeners();
  }

  /// 重新练习
  void retryPractice() {
    enterPractice();
  }

  @override
  void dispose() {
    AudioRecorderService.dispose();
    super.dispose();
  }
}
```

```dart
/// 将单词按音节展示，标注重音和音标，
/// 支持点击单个音节播放读音。
class SyllableBreakdown extends StatelessWidget {
  final WordSyllableInfo syllableInfo;
  final List<double>? syllableScores; // 可选：评测后的各音节得分
  final ValueChanged<int>? onSyllableTap;

  const SyllableBreakdown({
    super.key,
    required this.syllableInfo,
    this.syllableScores,
    this.onSyllableTap,
  });

  @override
  Widget build(BuildContext context) {
    return Column(
      children: [
        // 完整音标
        Text(
          syllableInfo.phonetic,
          style: TextStyle(
            fontSize: 18,
            color: Colors.grey.shade700,
            fontStyle: FontStyle.italic,
          ),
        ),
        const SizedBox(height: 16),

        // 音节拆解
        Row(
          mainAxisAlignment: MainAxisAlignment.center,
          children: syllableInfo.syllables.asMap().entries.map((entry) {
            final index = entry.key;
            final syllable = entry.value;
            final score = syllableScores != null && index < syllableScores!.length
                ? syllableScores![index]
                : null;

            return _SyllableChip(
              syllable: syllable,
              score: score,
              isLast: index == syllableInfo.syllables.length - 1,
              onTap: onSyllableTap != null
                  ? () => onSyllableTap!(index)
                  : null,
            );
          }).toList(),
        ),

        const SizedBox(height: 12),

        // 音节数量和重音说明
        Text(
          '${syllableInfo.syllableCount} 个音节  •  '
          '重音模式: ${_describeStress(syllableInfo)}',
          style: TextStyle(fontSize: 13, color: Colors.grey.shade600),
        ),
      ],
    );
  }

  String _describeStress(WordSyllableInfo info) {
    final stressIndex = info.syllables
        .indexWhere((s) => s.isPrimaryStress);
    if (stressIndex >= 0) {
      return '第${stressIndex + 1}音节重读';
    }
    return '均匀读';
  }
}

class _SyllableChip extends StatelessWidget {

                            style: TextStyle(
                              fontSize: 16,
                              fontWeight: FontWeight.bold,
                              color: Color(0xFF00C897),
                            ),

                        syllable.text,
                        style: TextStyle(
                          fontSize: 24,
                          fontWeight: syllable.isPrimaryStress
                              ? FontWeight.w800
                              : FontWeight.w500,
                          color: syllable.isPrimaryStress
                              ? const Color(0xFF00C897)
                              : Colors.black87,
                          letterSpacing: 1,
                        ),
                      ),
                    ],
                  ),
                ),

                const SizedBox(height: 6),

                // 音标
                Text(
                  '/${syllable.phoneme}/',
                  style: TextStyle(
                    fontSize: 14,
                    color: Colors.grey.shade600,
                    fontStyle: FontStyle.italic,

                // 评分（如有）

                  const SizedBox(height: 4),
```

```dart
class _PronunciationButtonState extends State<PronunciationButton>
    with SingleTickerProviderStateMixin {
  late AnimationController _pulseController;
  late Animation<double> _pulseAnimation;

  @override
  void initState() {
    super.initState();
    _pulseController = AnimationController(
      vsync: this,
      duration: const Duration(milliseconds: 1000),
    );
    _pulseAnimation = Tween<double>(begin: 1.0, end: 1.3).animate(
      CurvedAnimation(parent: _pulseController, curve: Curves.easeInOut),
    );
  }

  @override
  void didUpdateWidget(covariant PronunciationButton oldWidget) {
    super.didUpdateWidget(oldWidget);
    if (widget.state == RecordingState.recording) {
      _pulseController.repeat(reverse: true);
    } else {
      _pulseController.stop();
      _pulseController.reset();
    }
  }

  @override
  void dispose() {
    _pulseController.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Column(
      children: [
        GestureDetector(
          onLongPressStart: (_) => widget.onStartRecording(),
          onLongPressEnd: (_) => widget.onStopRecording(),
          onLongPressCancel: widget.onStopRecording,
          child: AnimatedBuilder(
            animation: _pulseAnimation,
            builder: (context, child) {
              final scale = widget.state == RecordingState.recording
                  ? _pulseAnimation.value
                  : 1.0;

              return Transform.scale(
                scale: scale,
                child: Container(
                  width: 80,
                  height: 80,
                  decoration: BoxDecoration(
                    shape: BoxShape.circle,
                    color: _getButtonColor(),
                    boxShadow: [
                      BoxShadow(
                        color: _getButtonColor().withOpacity(0.3),
                        blurRadius: widget.state == RecordingState.recording
                            ? 20
                            : 8,
                        spreadRadius: widget.state == RecordingState.recording
                            ? 5
                            : 0,
                      ),
                    ],
                  ),
                  child: _buildIcon(),
                ),
              );
            },
          ),
        ),

        const SizedBox(height: 12),

        // 状态提示文字
        Text(
          _getHintText(),
          style: TextStyle(
            fontSize: 14,
            color: Colors.grey.shade600,
          ),
        ),
      ],
    );
  }

  Widget _buildIcon() {
    switch (widget.state) {
      case RecordingState.idle:
        return const Icon(Icons.mic, color: Colors.white, size: 36);
      case RecordingState.recording:
        return const Icon(Icons.mic, color: Colors.white, size: 40);
      case RecordingState.processing:
        return const SizedBox(
          width: 36,
          height: 36,
          child: CircularProgressIndicator(
            color: Colors.white,
            strokeWidth: 3,
          ),
        );
      case RecordingState.evaluated:
        return const Icon(Icons.check, color: Colors.white, size: 36);
    }
  }

  Color _getButtonColor() {
    switch (widget.state) {
      case RecordingState.idle:
        return const Color(0xFF00C897);
      case RecordingState.recording:
        return const Color(0xFFFF4D4F);
      case RecordingState.processing:
        return Colors.orange;
      case RecordingState.evaluated:
        return const Color(0xFF00C897);
    }
  }

  String _getHintText() {

      case RecordingState.recording:

      case RecordingState.processing:
        return '正在评测...';
      case RecordingState.evaluated:
        return '评测完成';
```

```dart
        const SizedBox(height: 40),

        Text(
          provider.currentWord.word,
          style: const TextStyle(
            fontSize: 36,
            fontWeight: FontWeight.bold,
          ),
        ),

        if (provider.syllableInfo != null) ...[
          const SizedBox(height: 8),
          Text(
            provider.syllableInfo!.phonetic,
            style: TextStyle(
              fontSize: 18,
              color: Colors.grey.shade600,
              fontStyle: FontStyle.italic,
            ),
          ),
        ],

        const SizedBox(height: 48),

        // 录音按钮
        PronunciationButton(
          state: provider.recordingState,
          onStartRecording: () => provider.startRecording(),
          onStopRecording: () => provider.stopAndEvaluate(),
        ),
      ],
    );
  }

  /// 结果视图
  Widget _buildResultView(
    BuildContext context,
    WordMemoryProvider provider,
  ) {
    if (provider.pronunciationResult == null) {
      return const Center(child: Text('暂无评测结果'));
    }

    return ScoreResultPanel(
      result: provider.pronunciationResult!,
      syllableInfo: provider.syllableInfo,
      onRetry: () => provider.retryPractice(),
      onNext: () {
        if (provider.hasNext) {
          provider.nextWord();
        } else {
          // 学完了
          _showCompletionDialog(context, provider);
        }
      },
    );
  }

  /// 底部导航栏
  Widget _buildBottomBar(
    BuildContext context,
    WordMemoryProvider provider,
  ) {
    return Container(
      padding: const EdgeInsets.all(16),
      decoration: BoxDecoration(
        color: Colors.white,
        boxShadow: [
          BoxShadow(
            color: Colors.black.withOpacity(0.05),
            blurRadius: 10,
            offset: const Offset(0, -2),
          ),
        ],
      ),
      child: SafeArea(
        child: Row(
          mainAxisAlignment: MainAxisAlignment.spaceBetween,
          children: [
            TextButton.icon(
              onPressed:
                  provider.hasPrevious ? () => provider.previousWord() : null,
              icon: const Icon(Icons.arrow_back),
              label: const Text('上一个'),
            ),

            // 步骤指示器
            Row(
              children: List.generate(4, (i) {
                final isActive = i <= provider.learningStep.index;
                return Container(
                  width: 8,
                  height: 8,
                  margin: const EdgeInsets.symmetric(horizontal: 3),
                  decoration: BoxDecoration(
                    shape: BoxShape.circle,
                    color: isActive
                        ? const Color(0xFF00C897)
                        : Colors.grey.shade300,
                  ),
                );
              }),
            ),

            TextButton.icon(
              onPressed:
                  provider.hasNext ? () => provider.nextWord() : null,
              icon: const Icon(Icons.arrow_forward),
              label: const Text('跳过'),
            ),
          ],
        ),
      ),
    );
  }

  void _showCompletionDialog(
    BuildContext context,
    WordMemoryProvider provider,
  ) {
    showDialog(
      context: context,
      builder: (ctx) => AlertDialog(
        title: const Text('🎉 恭喜完成！'),
        content: Text(
          '你已经完成了全部 ${provider.totalWords} 个单词的学习！',
        ),
        actions: [
          TextButton(
            onPressed: () {
              Navigator.pop(ctx);
              provider.goToWord(0); // 重新开始
            },
            child: const Text('重新学习'),
          ),
          ElevatedButton(
            onPressed: () {
              Navigator.pop(ctx);
              Navigator.pop(context); // 返回上一页
            },
            child: const Text('完成'),
          ),
        ],
      ),
    );
  }
}
```
