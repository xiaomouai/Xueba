// -------------------- 发音功能 --------------------
const voiceSelect = document.getElementById('voiceAccent');
const rateEl = document.getElementById('rate');
const pitchEl = document.getElementById('pitch');
const stopBtn = document.getElementById('stopBtn');

let synth = window.speechSynthesis;
let availableVoices = [];

function loadVoices(){
  availableVoices = synth.getVoices() || [];
}
loadVoices();
if (typeof speechSynthesis !== 'undefined'){
  speechSynthesis.onvoiceschanged = loadVoices;
}

function pickVoiceForLocale(localePrefix){
  // 尝试选取合适的 voice（优先有 locale 前缀的）
  if (!availableVoices.length) loadVoices();
  let v = availableVoices.find(x => x.lang && x.lang.startsWith(localePrefix) && x.name.toLowerCase().includes('female')) 
        || availableVoices.find(x => x.lang && x.lang.startsWith(localePrefix));
  return v || availableVoices[0] || null;
}

function stopSpeaking(){
  if (synth.speaking) synth.cancel();
}
stopBtn.addEventListener('click', stopSpeaking);

// 播放项（优先尝试 item.audio，如果为空则使用合成语音）
function playItem(item){
  stopSpeaking();
  // 如果存在外部音频 URL（item.audio），优先用 audio 标签播放（适合高质量录音替换）。
  if (item.audio){
    const a = new Audio(item.audio);
    a.play();
    return;
  }

  // 否则使用 SpeechSynthesis 播放示例词（speakText）或 symbol 本身
  const text = item.speakText || item.example || item.symbol;
  const locale = voiceSelect.value || 'en-GB';
  const voice = pickVoiceForLocale(locale) || undefined;

  const u = new SpeechSynthesisUtterance(text);
  if (voice) u.voice = voice;
  u.lang = locale;
  u.rate = parseFloat(rateEl.value) || 1;
  u.pitch = parseFloat(pitchEl.value) || 1;

  // 为了更贴近单个音素的发音：若 item.symbol 看起来像纯元音（含 / 或是单字母如 iː），
  // 我们优先让语音读示例词中的单音节（speakText），并且把 volume 设为 1。
  u.onend = ()=>{/* 可扩展：播放完成后动作 */};
  synth.speak(u);
}
