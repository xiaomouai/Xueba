### services/level_service.py

```python
"""等级服务"""
from typing import Optional, Tuple
from sqlalchemy.orm import Session
from sqlalchemy import asc
from models.checkin import LevelConfig
from models.user import User


class LevelService:
    def __init__(self, db: Session):
        self.db = db
        self._level_cache: list = []

    def _load_levels(self):
        if not self._level_cache:
            self._level_cache = (
                self.db.query(LevelConfig)
                .order_by(asc(LevelConfig.level))
                .all()
            )
        return self._level_cache

    def get_user_level_info(self, user: User) -> dict:
        """获取用户等级信息"""
        levels = self._load_levels()
        current = None
        next_level = None

        for i, lv in enumerate(levels):
            if lv.level == user.level:
                current = lv
                if i + 1 < len(levels):
                    next_level = levels[i + 1]
                break

        if not current:
            current = levels[0]

        # 计算进度
        if next_level:
            range_points = next_level.min_points - current.min_points
            user_progress = user.total_points - current.min_points
            progress = min(max(user_progress / range_points, 0.0), 1.0)
        else:
            progress = 1.0  # 满级

        return {
            "level": current.level,
            "level_name": current.level_name,
            "icon": current.icon,
            "color": current.color,
            "description": current.description,
            "min_points": current.min_points,
            "next_level_points": next_level.min_points if next_level else None,
            "progress": round(progress, 4),
        }

    def get_daily_base_points(self, level: int) -> int:
        """获取该等级每日基础签到积分"""
        levels = self._load_levels()
        for lv in levels:
            if lv.level == level:
                return lv.daily_base_points
        return 10

    def check_level_up(self, user: User) -> Tuple[bool, Optional[int]]:
        """
        检查是否升级
        返回: (是否升级, 新等级)
        """
        levels = self._load_levels()
        new_level = user.level

        for lv in levels:
            if user.total_points >= lv.min_points:
                new_level = lv.level

        if new_level > user.level:
            return True, new_level
        return False, None
```

```python
"""积分服务"""
from sqlalchemy.orm import Session
from models.user import User
from models.points import PointsLog


class PointsService:
    def __init__(self, db: Session):
        self.db = db

    def add_points(
        self,
        user: User,
        points: int,
        change_type: str,
        description: str,
        ref_id: int = None,
    ):
        """增加积分"""
        before = user.available_points

        user.available_points += points
        user.total_points += points

        log = PointsLog(
            user_id=user.id,
            change_type=change_type,
            change_points=points,
            before_points=before,
            after_points=user.available_points,
            description=description,
            ref_id=ref_id,
        )
        self.db.add(log)

    def consume_points(
        self,
        user: User,
        points: int,
        change_type: str,
        description: str,
        ref_id: int = None,
    ) -> bool:
        """消费积分"""
        if user.available_points < points:
            return False

        before = user.available_points
        user.available_points -= points

        log = PointsLog(
            user_id=user.id,
            change_type=change_type,
            change_points=-points,
            before_points=before,
            after_points=user.available_points,
            description=description,
            ref_id=ref_id,
        )
        self.db.add(log)
        return True

    def get_points_history(self, user_id: int, page: int = 1, size: int = 20):
        """获取积分流水"""
        query = (
            self.db.query(PointsLog)
            .filter(PointsLog.user_id == user_id)
            .order_by(PointsLog.created_at.desc())
        )
        total = query.count()
        records = query.offset((page - 1) * size).limit(size).all()

        return {
            "total": total,
            "page": page,
            "size": size,
            "records": [
                {
                    "change_type": r.change_type,
                    "change_points": r.change_points,
                    "description": r.description,
                    "created_at": r.created_at.isoformat(),
                }
                for r in records
            ],
        }
```

### services/checkin_service.py

```python
"""签到核心服务"""
from datetime import date, datetime, timedelta
from typing import List, Optional
from sqlalchemy.orm import Session
from sqlalchemy import and_, extract

from models.user import User
from models.checkin import CheckinRecord, ConsecutiveRewardConfig
from models.points import UserReward
from services.points_service import PointsService
from services.level_service import LevelService


class CheckinService:
    def __init__(self, db: Session):
        self.db = db
        self.points_service = PointsService(db)
        self.level_service = LevelService(db)

    # ──────────────────────────────────────────
    # 签到
    # ──────────────────────────────────────────
    def do_checkin(self, user_id: int) -> dict:
        """执行签到"""
        today = date.today()
        user = self.db.query(User).filter(User.id == user_id).with_for_update().first()
        if not user:
            return {"success": False, "message": "用户不存在"}

        # 1) 检查今日是否已签到
        existing = (
            self.db.query(CheckinRecord)
            .filter(and_(
                CheckinRecord.user_id == user_id,
                CheckinRecord.checkin_date == today,
            ))
            .first()
        )
        if existing:
            return {
                "success": False,
                "message": "今日已签到",
                "is_checked_today": True,
                "consecutive_days": user.consecutive_days,
            }

        # 2) 计算连续签到天数
        yesterday = today - timedelta(days=1)
        if user.last_checkin_date == yesterday:
            consecutive = user.consecutive_days + 1
        elif user.last_checkin_date == today:
            consecutive = user.consecutive_days  # 不应该走到这里
        else:
            consecutive = 1  # 中断, 重新计数

        # 3) 计算积分
        base_points = self.level_service.get_daily_base_points(user.level)

        # 连续签到额外加成: 每连续 1 天 +1 积分, 上限 +20
        streak_bonus = min(consecutive - 1, 20)
        bonus_points = streak_bonus

        # 4) 检查连续签到里程碑奖励
        rewards_earned = self._check_consecutive_rewards(user, consecutive)
        milestone_bonus = sum(r["reward_value"] for r in rewards_earned)
        bonus_points += milestone_bonus

        total_earned = base_points + bonus_points

        # 5) 创建签到记录
        record = CheckinRecord(
            user_id=user_id,
            checkin_date=today,
            consecutive_day=consecutive,
            base_points=base_points,
            bonus_points=bonus_points,
            total_points=total_earned,
        )
        self.db.add(record)
        self.db.flush()

        # 6) 加积分
        self.points_service.add_points(
            user=user,
            points=total_earned,
            change_type="checkin",
            description=f"每日签到(连续第{consecutive}天)",
            ref_id=record.id,
        )

        # 7) 更新用户签到信息
        user.last_checkin_date = today
        user.consecutive_days = consecutive
        user.total_checkin_days += 1
        if consecutive > user.max_consecutive_days:
            user.max_consecutive_days = consecutive

        # 8) 检查是否升级
        level_up, new_level = self.level_service.check_level_up(user)
        if level_up and new_level:
            user.level = new_level

        self.db.commit()

        return {
            "success": True,
            "message": "签到成功！",
            "is_checked_today": True,
            "checkin_points": base_points,
            "bonus_points": bonus_points,
            "total_earned": total_earned,
            "consecutive_days": consecutive,
            "rewards_earned": rewards_earned,
            "level_up": level_up,
            "new_level": new_level,
            "total_points": user.total_points,
            "available_points": user.available_points,
        }

    # ──────────────────────────────────────────
    # 连续签到里程碑奖励
    # ──────────────────────────────────────────
    def _check_consecutive_rewards(self, user: User, consecutive_days: int) -> list:
        """检查并发放连续签到奖励"""
        rewards_earned = []

        configs = (
            self.db.query(ConsecutiveRewardConfig)
            .filter(ConsecutiveRewardConfig.consecutive_days == consecutive_days)
            .all()
        )

        for cfg in configs:
            # 检查是否已获得过(非可重复的)
            if not cfg.is_repeatable:
                already = (
                    self.db.query(UserReward)
                    .filter(and_(
                        UserReward.user_id == user.id,
                        UserReward.reward_config_id == cfg.id,
                    ))
                    .first()
                )
                if already:
                    continue

            # 发放奖励
            reward = UserReward(
                user_id=user.id,
                reward_config_id=cfg.id,
                consecutive_days=consecutive_days,
                reward_type=cfg.reward_type,
                reward_value=cfg.reward_value,
                reward_name=cfg.reward_name,
            )
            self.db.add(reward)

            rewards_earned.append({
                "consecutive_days": cfg.consecutive_days,
                "reward_name": cfg.reward_name,
                "reward_icon": cfg.reward_icon,
                "reward_value": cfg.reward_value,
                "reward_type": cfg.reward_type,
                "description": cfg.description,
            })

        return rewards_earned

    # ──────────────────────────────────────────
    # 签到页面数据
    # ──────────────────────────────────────────
    def get_checkin_page_data(self, user_id: int, year: int = None, month: int = None) -> dict:
        """获取签到页面所需的全部数据"""
        today = date.today()
        year = year or today.year
        month = month or today.month

        user = self.db.query(User).filter(User.id == user_id).first()
        if not user:
            return {"error": "用户不存在"}

        # 1) 月度签到日历
        calendar_days = self._get_month_calendar(user_id, year, month)

        # 2) 等级信息
        level_info = self.level_service.get_user_level_info(user)

        # 3) 连续签到奖励进度
        rewards = self._get_consecutive_rewards_progress(user)

        # 4) 今日是否已签到
        is_checked_today = user.last_checkin_date == today

        return {
            "user_id": user.id,
            "username": user.username,
            "avatar_url": user.avatar_url,
            "level_info": level_info,
            "total_points": user.total_points,
            "available_points": user.available_points,
            "total_checkin_days": user.total_checkin_days,
            "consecutive_days": user.consecutive_days,
            "max_consecutive_days": user.max_consecutive_days,
            "is_checked_today": is_checked_today,
            "calendar_days": calendar_days,
            "consecutive_rewards": rewards,
        }

    def _get_month_calendar(self, user_id: int, year: int, month: int) -> list:
        """获取某月签到记录"""
        records = (
            self.db.query(CheckinRecord)
            .filter(and_(
                CheckinRecord.user_id == user_id,
                extract("year", CheckinRecord.checkin_date) == year,
                extract("month", CheckinRecord.checkin_date) == month,
            ))
            .all()
        )

        checked_map = {
            r.checkin_date: {
                "points": r.total_points,
                "consecutive_day": r.consecutive_day,
            }
            for r in records
        }

        # 生成当月每一天
        import calendar
        _, days_in_month = calendar.monthrange(year, month)
        result = []
        for day in range(1, days_in_month + 1):
            d = date(year, month, day)
            info = checked_map.get(d)
            result.append({
                "date": d.isoformat(),
                "is_checked": info is not None,
                "points": info["points"] if info else 0,
                "consecutive_day": info["consecutive_day"] if info else 0,
            })

        return result

    def _get_consecutive_rewards_progress(self, user: User) -> list:
        """获取连续签到奖励进度"""
        configs = (
            self.db.query(ConsecutiveRewardConfig)
            .order_by(ConsecutiveRewardConfig.consecutive_days.asc())
            .all()
        )

        # 用户已获得的奖励
        claimed_ids = set()
        claimed = (
            self.db.query(UserReward)
            .filter(UserReward.user_id == user.id)
            .all()
        )
        for c in claimed:
            claimed_ids.add(c.reward_config_id)

        result = []
        current_target_found = False

        for cfg in configs:
            is_claimed = cfg.id in claimed_ids
            is_current_target = (
                not current_target_found
                and not is_claimed
                and user.consecutive_days < cfg.consecutive_days
            )
            if is_current_target:
                current_target_found = True

            result.append({
                "consecutive_days": cfg.consecutive_days,
                "reward_name": cfg.reward_name,
                "reward_icon": cfg.reward_icon,
                "reward_value": cfg.reward_value,
                "reward_type": cfg.reward_type,
                "description": cfg.description,
                "is_claimed": is_claimed,
                "is_current_target": is_current_target,
            })

        return result

    # ──────────────────────────────────────────
    # 签到排行榜
    # ──────────────────────────────────────────
    def get_leaderboard(self, limit: int = 20) -> list:
        """签到排行榜"""
        users = (
            self.db.query(User)
            .order_by(User.total_checkin_days.desc(), User.consecutive_days.desc())
            .limit(limit)
            .all()
        )
        return [
            {
                "rank": i + 1,
                "user_id": u.id,
                "username": u.username,
                "avatar_url": u.avatar_url,
                "level": u.level,
                "total_checkin_days": u.total_checkin_days,
                "consecutive_days": u.consecutive_days,
                "total_points": u.total_points,
            }
            for i, u in enumerate(users)
        ]
```

```dart
import 'dart:convert';
import 'package:http/http.dart' as http;
import '../models/api_response.dart';
import '../models/checkin_models.dart';
import '../utils/constants.dart';

class CheckinApiService {
  static final _client = http.Client();

  /// 获取签到页面数据
  static Future<CheckinPageData?> getPageData(int userId, {int? year, int? month}) async {
    try {
      final params = {'user_id': userId.toString()};
      if (year != null) params['year'] = year.toString();
      if (month != null) params['month'] = month.toString();

      final uri = Uri.parse('${ApiConstants.baseUrl}${ApiConstants.checkinPageData}')
          .replace(queryParameters: params);

      final response = await _client.get(uri).timeout(const Duration(seconds: 10));
      final api = ApiResponse.fromJson(jsonDecode(response.body));

      if (api.isSuccess && api.data != null) {
        return CheckinPageData.fromJson(api.data);
      }
    } catch (e) {
      print('获取签到页面数据失败: $e');
    }
    return null;
  }

  /// 执行签到
  static Future<CheckinResult> doCheckin(int userId) async {
    try {
      final response = await _client.post(
        Uri.parse('${ApiConstants.baseUrl}${ApiConstants.checkinDo}'),
        headers: {'Content-Type': 'application/json'},
        body: jsonEncode({'user_id': userId}),
      ).timeout(const Duration(seconds: 10));

      final api = ApiResponse.fromJson(jsonDecode(response.body));

      if (api.isSuccess && api.data != null) {
        return CheckinResult.fromJson(api.data);
      }
      return CheckinResult(
        success: false,
        message: api.message,
      );
    } catch (e) {
      return CheckinResult(success: false, message: '网络错误，请稍后重试');
    }
  }
}
```

```dart
class _ParticlePainter extends CustomPainter {
  final double progress;
  final int particleCount = 12;
  final Random _random = Random(42);

  _ParticlePainter({required this.progress});

  @override
  void paint(Canvas canvas, Size size) {
    final center = Offset(size.width / 2, size.height / 2);

    for (int i = 0; i < particleCount; i++) {
      final angle = (i / particleCount) * pi * 2;
      final distance = 40 + progress * 60;
      final opacity = (1 - progress).clamp(0.0, 1.0);
      final particleSize = (4 * (1 - progress)).clamp(0.0, 4.0);

      final colors = [Colors.orange, Colors.red, Colors.yellow, Colors.amber];
      final paint = Paint()
        ..color = colors[i % colors.length].withOpacity(opacity)
        ..style = PaintingStyle.fill;

      final x = center.dx + cos(angle) * distance;
      final y = center.dy + sin(angle) * distance;

      canvas.drawCircle(Offset(x, y), particleSize, paint);
    }
  }

  @override
  bool shouldRepaint(covariant _ParticlePainter oldDelegate) {
    return oldDelegate.progress != progress;
  }
}
```

```dart
class PointsEarnedAnimation extends StatefulWidget {
  final int points;
  final VoidCallback? onComplete;

  const PointsEarnedAnimation({
    super.key,
    required this.points,
    this.onComplete,
  });

  @override
  State<PointsEarnedAnimation> createState() => _PointsEarnedAnimationState();
}

class _PointsEarnedAnimationState extends State<PointsEarnedAnimation>
    with SingleTickerProviderStateMixin {
  late AnimationController _controller;
  late Animation<double> _fadeAnimation;
  late Animation<Offset> _slideAnimation;

  @override
  void initState() {
    super.initState();
    _controller = AnimationController(
      vsync: this,
      duration: const Duration(milliseconds: 1500),
    );

    _fadeAnimation = Tween<double>(begin: 1.0, end: 0.0).animate(
      CurvedAnimation(
        parent: _controller,
        curve: const Interval(0.6, 1.0, curve: Curves.easeOut),
      ),
    );

    _slideAnimation = Tween<Offset>(
      begin: Offset.zero,
      end: const Offset(0, -2),
    ).animate(
      CurvedAnimation(parent: _controller, curve: Curves.easeOut),
    );

    _controller.forward().then((_) {
      widget.onComplete?.call();
    });
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return SlideTransition(
      position: _slideAnimation,
      child: FadeTransition(
        opacity: _fadeAnimation,
        child: Text(
          '+${widget.points}',
          style: const TextStyle(
            fontSize: 28,
            fontWeight: FontWeight.bold,
            color: Colors.orange,
            shadows: [
              Shadow(color: Colors.orangeAccent, blurRadius: 10),
            ],
          ),
        ),
      ),
    );
  }
}
```

```dart
  State<CheckinScreen> createState() => _CheckinScreenState();

class _CheckinScreenState extends State<CheckinScreen> {
  bool _showPointsAnimation = false;
  int _earnedPoints = 0;

  @override
  void initState() {
    super.initState();
    WidgetsBinding.instance.addPostFrameCallback((_) {
      context.read<CheckinProvider>().loadPageData(widget.userId);
    });
  }

  Future<void> _handleCheckin() async {
    final provider = context.read<CheckinProvider>();
    final result = await provider.doCheckin(widget.userId);

    if (!mounted) return;

    if (result.success) {
      // 显示积分飘字
      setState(() {
        _showPointsAnimation = true;
        _earnedPoints = result.totalEarned;
      });

      // 弹出奖励弹窗
      await Future.delayed(const Duration(milliseconds: 500));
      if (mounted) {
        RewardPopup.showCheckinSuccess(context, result);
      }
    } else {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text(result.message),
          behavior: SnackBarBehavior.floating,
          shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(10)),
        ),
      );
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: const Color(0xFFF5F6FA),
      appBar: AppBar(
        title: const Text('每日签到'),
        centerTitle: true,
        elevation: 0,
        backgroundColor: Colors.transparent,
        foregroundColor: Colors.black87,
        actions: [
          TextButton.icon(
            onPressed: () {
              // TODO: 跳转积分明细页
            },
            icon: const Icon(Icons.history, size: 18),
            label: const Text('明细'),
          ),
        ],
      ),
      body: Consumer<CheckinProvider>(
        builder: (context, provider, _) {
          if (provider.isLoading && provider.pageData == null) {
            return const Center(child: CircularProgressIndicator());
          }

          final data = provider.pageData;
          if (data == null) {
            return Center(
              child: Column(
                mainAxisSize: MainAxisSize.min,
                children: [
                  const Text('加载失败'),
                  const SizedBox(height: 12),
                  ElevatedButton(
                    onPressed: () => provider.loadPageData(widget.userId),
                    child: const Text('重试'),
                  ),
                ],
              ),
            );
          }

          return RefreshIndicator(
            onRefresh: () => provider.loadPageData(widget.userId),
            child: SingleChildScrollView(
              physics: const AlwaysScrollableScrollPhysics(),
              child: Column(
                children: [
                  const SizedBox(height: 8),

                  // ① 等级与积分卡片
                  LevelProgressCard(
                    levelInfo: data.levelInfo,
                    totalPoints: data.totalPoints,
                    availablePoints: data.availablePoints,
                    consecutiveDays: data.consecutiveDays,
                    totalCheckinDays: data.totalCheckinDays,
                  ),
                  const SizedBox(height: 24),

                  // ② 签到按钮 + 积分飘字
                  Stack(
                    alignment: Alignment.center,
                    children: [
                      CheckinButton(
                        isChecked: data.isCheckedToday,
                        isLoading: provider.isChecking,
                        consecutiveDays: data.consecutiveDays,
                        onPressed: _handleCheckin,
                      ),
                      if (_showPointsAnimation)
                        Positioned(
                          top: 0,
                          child: PointsEarnedAnimation(
                            points: _earnedPoints,
                            onComplete: () {
                              if (mounted) {
                                setState(() => _showPointsAnimation = false);
                              }
                            },

                        ),
                    ],
                  ),
                  const SizedBox(height: 24),

                  // ③ 连续签到奖励进度

                      int y = provider.currentYear;
                      int m = provider.currentMonth - 1;
                      if (m < 1) { m = 12; y--; }
                      provider.changeMonth(widget.userId, y, m);
                    },
                    onNextMonth: () {
                      int y = provider.currentYear;
                      int m = provider.currentMonth + 1;
                      if (m > 12) { m = 1; y++; }
                      provider.changeMonth(widget.userId, y, m);
                    },
                  ),
                  const SizedBox(height: 32),
                ],
              ),
            ),
          );
        },
      ),
    );
  }
```
