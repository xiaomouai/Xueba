### services/code_generator.py

```python
"""激活码生成器"""
import secrets
import string
import hashlib
from config import settings

class CodeGenerator:
    """激活码生成器"""

    @staticmethod
    def generate(prefix: str = None, length: int = None) -> str:
        """
        生成激活码
        格式: ACT-XXXX-XXXX-XXXX-XXXX
        """
        prefix = prefix or settings.CODE_PREFIX
        length = length or settings.CODE_LENGTH

        # 使用安全随机数生成
        chars = string.ascii_uppercase + string.digits
        # 排除容易混淆的字符
        chars = chars.replace('O', '').replace('0', '').replace('I', '').replace('1', '').replace('L', '')

        raw_code = ''.join(secrets.choice(chars) for _ in range(length))

        # 格式化: ACT-XXXX-XXXX-XXXX-XXXX
        formatted = '-'.join([raw_code[i:i+4] for i in range(0, len(raw_code), 4)])
        code = f"{prefix}-{formatted}"

        return code

    @staticmethod
    def generate_batch(count: int, prefix: str = None) -> list:
        """批量生成激活码"""
        codes = set()
        while len(codes) < count:
            codes.add(CodeGenerator.generate(prefix))
        return list(codes)

    @staticmethod
    def normalize_code(code: str) -> str:
        """标准化激活码(去除空格,统一大写)"""
        return code.strip().upper().replace(" ", "")
```

### services/activation_service.py

```python
"""激活服务核心逻辑"""
from datetime import datetime, timedelta
from typing import Optional, Tuple
from sqlalchemy.orm import Session
from sqlalchemy import and_

from models.activation_code import ActivationCode
from models.user import User
from models.user_license import UserLicense, ActivationLog
from services.code_generator import CodeGenerator
from config import settings


class ActivationService:
    """激活服务"""

    def __init__(self, db: Session):
        self.db = db

    def generate_codes(
        self,
        order_id: str,
        plan_type: str,
        quantity: int = 1,
        max_devices: int = 1
    ) -> list:
        """
        生成激活码(第三方下单后调用)
        """
        if plan_type not in settings.PLAN_DURATION:
            raise ValueError(f"无效的套餐类型: {plan_type}")

        duration_days = settings.PLAN_DURATION[plan_type]
        codes = CodeGenerator.generate_batch(quantity)
        result = []

        for code in codes:
            # 确保唯一性
            existing = self.db.query(ActivationCode).filter(
                ActivationCode.code == code
            ).first()
            if existing:
                code = CodeGenerator.generate()  # 重新生成

            activation_code = ActivationCode(
                code=code,
                order_id=order_id,
                plan_type=plan_type,
                duration_days=duration_days,
                status=0,
                max_devices=max_devices,
                expired_at=datetime.now() + timedelta(days=settings.CODE_VALIDITY_DAYS)
            )
            self.db.add(activation_code)
            result.append({
                "code": code,
                "plan_type": plan_type,
                "duration_days": duration_days
            })

        self.db.commit()
        return result

    def activate(
        self,
        code: str,
        device_id: str,
        ip_address: str = None,
        email: str = None,
        phone: str = None
    ) -> Tuple[bool, str, Optional[dict]]:
        """
        激活操作
        返回: (是否成功, 消息, 数据)
        """
        # 1. 标准化激活码
        normalized_code = CodeGenerator.normalize_code(code)

        # 2. 查找激活码
        activation_code = self.db.query(ActivationCode).filter(
            ActivationCode.code == normalized_code
        ).with_for_update().first()  # 加行锁防止并发

        # 3. 验证激活码
        if not activation_code:
            self._log_activation(normalized_code, device_id, ip_address, False, "激活码不存在")
            return False, "激活码不存在，请检查输入是否正确", None

        if activation_code.status == 1:
            self._log_activation(normalized_code, device_id, ip_address, False, "激活码已被使用")
            return False, "该激活码已被使用", None

        if activation_code.status == 2:
            self._log_activation(normalized_code, device_id, ip_address, False, "激活码已过期")
            return False, "该激活码已过期", None

        if activation_code.status == 3:
            self._log_activation(normalized_code, device_id, ip_address, False, "激活码已作废")
            return False, "该激活码已作废", None

        # 检查激活码本身是否过期
        if activation_code.expired_at and activation_code.expired_at < datetime.now():
            activation_code.status = 2
            self.db.commit()
            self._log_activation(normalized_code, device_id, ip_address, False, "激活码已过期")
            return False, "该激活码已过期，请联系客服", None

        # 4. 获取或创建用户
        user = self._get_or_create_user(device_id, email, phone)

        # 5. 检查是否已有有效许可证(可选: 允许叠加或拒绝)
        existing_license = self.db.query(UserLicense).filter(
            and_(
                UserLicense.user_id == user.id,
                UserLicense.device_id == device_id,
                UserLicense.status == 1,
                UserLicense.expires_at > datetime.now()
            )
        ).first()

        # 6. 计算过期时间
        now = datetime.now()
        if existing_license:
            # 如果有现有许可证，在现有基础上叠加
            base_time = existing_license.expires_at
        else:
            base_time = now

        expires_at = base_time + timedelta(days=activation_code.duration_days)

        # 7. 创建许可证
        license_record = UserLicense(
            user_id=user.id,
            activation_code_id=activation_code.id,
            device_id=device_id,
            plan_type=activation_code.plan_type,
            activated_at=now,
            expires_at=expires_at,
            status=1
        )
        self.db.add(license_record)

        # 8. 更新激活码状态
        activation_code.status = 1
        activation_code.activated_at = now
        activation_code.activated_by = user.id
        activation_code.activated_device_id = device_id

        # 9. 如果有现有许可证，更新其过期时间
        if existing_license:
            existing_license.expires_at = expires_at

        self.db.commit()

        # 10. 记录日志
        self._log_activation(normalized_code, device_id, ip_address, True, None)

        # 11. 返回结果
        remaining_days = (expires_at - now).days
        result_data = {
            "license_id": license_record.id,
            "plan_type": activation_code.plan_type,
            "activated_at": now.isoformat(),
            "expires_at": expires_at.isoformat(),
            "remaining_days": remaining_days,
            "is_active": True
        }

        return True, "激活成功！", result_data

    def verify_license(self, device_id: str) -> Tuple[bool, Optional[dict]]:
        """
        验证设备许可证状态
        """
        license_record = self.db.query(UserLicense).filter(
            and_(
                UserLicense.device_id == device_id,
                UserLicense.status == 1
            )
        ).order_by(UserLicense.expires_at.desc()).first()

        if not license_record:
            return False, None

        now = datetime.now()

        # 检查是否已过期
        if license_record.expires_at < now:
            license_record.status = 0  # 标记为过期
            self.db.commit()
            return False, None

        remaining_days = (license_record.expires_at - now).days
        return True, {
            "license_id": license_record.id,
            "plan_type": license_record.plan_type,
            "activated_at": license_record.activated_at.isoformat(),
            "expires_at": license_record.expires_at.isoformat(),
            "remaining_days": remaining_days,
            "is_active": True
        }

    def _get_or_create_user(
        self,
        device_id: str,
        email: str = None,
        phone: str = None
    ) -> User:
        """获取或创建用户"""
        user = self.db.query(User).filter(
            User.device_id == device_id
        ).first()

        if not user:
            user = User(
                device_id=device_id,
                email=email,
                phone=phone,
                status=1
            )
            self.db.add(user)
            self.db.flush()  # 获取ID
        else:
            # 更新信息
            if email:
                user.email = email
            if phone:
                user.phone = phone

        return user

    def _log_activation(
        self,
        code: str,
        device_id: str,
        ip_address: str,
        success: bool,
        failure_reason: str = None
    ):
        """记录激活日志"""
        log = ActivationLog(
            activation_code=code,
            device_id=device_id,
            ip_address=ip_address,
            result=1 if success else 0,
            failure_reason=failure_reason
        )
        self.db.add(log)
        self.db.commit()
```

### services/license_service.py

```python
"""许可证验证服务"""
import json
import hashlib
import hmac
from datetime import datetime
from typing import Optional
from config import settings


class LicenseTokenService:
    """
    生成本地验证Token,减少网络请求
    客户端可以缓存Token,定期校验
    """

    @staticmethod
    def generate_token(license_data: dict) -> str:
        """生成许可证Token"""
        payload = json.dumps(license_data, sort_keys=True)
        signature = hmac.new(
            settings.SECRET_KEY.encode(),
            payload.encode(),
            hashlib.sha256
        ).hexdigest()

        import base64
        token_data = {
            "payload": license_data,
            "signature": signature,
            "generated_at": datetime.now().isoformat()
        }
        token = base64.b64encode(json.dumps(token_data).encode()).decode()
        return token

    @staticmethod
    def verify_token(token: str) -> Optional[dict]:
        """验证许可证Token"""
        try:
            import base64
            token_data = json.loads(base64.b64decode(token.encode()).decode())

            payload = json.dumps(token_data["payload"], sort_keys=True)
            expected_signature = hmac.new(
                settings.SECRET_KEY.encode(),
                payload.encode(),
                hashlib.sha256
            ).hexdigest()

            if not hmac.compare_digest(expected_signature, token_data["signature"]):
                return None

            # 检查是否过期
            license_data = token_data["payload"]
            expires_at = datetime.fromisoformat(license_data["expires_at"])
            if expires_at < datetime.now():
                return None

            return license_data
        except Exception:
            return None
```

```python
    激活接口

        phone=body.phone

    # 生成本地验证Token
    if data:
        data["license_token"] = LicenseTokenService.generate_token(data)
```

```python
async def get_license_status(
    device_id: str = Query(..., description="设备ID"),
    db: Session = Depends(get_db)
):
    """获取设备的许可证状态"""
    license_record = db.query(UserLicense).filter(
        and_(
            UserLicense.device_id == device_id,
            UserLicense.status == 1
        )
    ).order_by(UserLicense.expires_at.desc()).first()

    if not license_record:
        return ApiResponse.success(data={
            "is_active": False,
            "needs_activation": True
        })

    now = datetime.now()
    if license_record.expires_at < now:
        license_record.status = 0
        db.commit()
        return ApiResponse.success(data={
            "is_active": False,
            "expired": True,
            "needs_activation": True
        })

    remaining_days = (license_record.expires_at - now).days
    return ApiResponse.success(data={
        "is_active": True,
        "plan_type": license_record.plan_type,
        "activated_at": license_record.activated_at.isoformat(),
        "expires_at": license_record.expires_at.isoformat(),
```

```dart
import 'dart:io';
import 'package:device_info_plus/device_info_plus.dart';
import 'package:flutter_secure_storage/flutter_secure_storage.dart';
import '../utils/constants.dart';

/// 设备服务 - 获取唯一设备标识
class DeviceService {
  static final DeviceInfoPlugin _deviceInfo = DeviceInfoPlugin();
  static final FlutterSecureStorage _secureStorage =
      const FlutterSecureStorage();

  /// 获取设备唯一ID
  static Future<String> getDeviceId() async {
    // 先从安全存储读取
    String? storedId = await _secureStorage.read(key: AppConstants.deviceIdKey);
    if (storedId != null && storedId.isNotEmpty) {
      return storedId;
    }

    // 生成设备ID
    String deviceId = await _generateDeviceId();

    // 保存到安全存储
    await _secureStorage.write(
      key: AppConstants.deviceIdKey,
      value: deviceId,
    );

    return deviceId;
  }

  /// 生成设备ID
  static Future<String> _generateDeviceId() async {
    try {
      if (Platform.isAndroid) {
        final androidInfo = await _deviceInfo.androidInfo;
        return 'android_${androidInfo.id}';
      } else if (Platform.isIOS) {
        final iosInfo = await _deviceInfo.iosInfo;
        return 'ios_${iosInfo.identifierForVendor ?? DateTime.now().millisecondsSinceEpoch}';
      } else if (Platform.isWindows) {
        final windowsInfo = await _deviceInfo.windowsInfo;
        return 'win_${windowsInfo.deviceId}';
      } else if (Platform.isMacOS) {
        final macInfo = await _deviceInfo.macOsInfo;
        return 'mac_${macInfo.systemGUID ?? DateTime.now().millisecondsSinceEpoch}';
      }
    } catch (e) {
      // fallback
    }
    return 'device_${DateTime.now().millisecondsSinceEpoch}';
  }

  /// 获取设备信息
  static Future<Map<String, String>> getDeviceInfo() async {
    try {
      if (Platform.isAndroid) {
        final info = await _deviceInfo.androidInfo;
        return {
          'platform': 'android',
          'model': info.model,
          'brand': info.brand,
          'version': info.version.release,
        };
      } else if (Platform.isIOS) {
        final info = await _deviceInfo.iosInfo;
        return {
          'platform': 'ios',
          'model': info.model,
          'name': info.name,
          'version': info.systemVersion,
        };
      }
    } catch (e) {
      // ignore
    }
    return {'platform': Platform.operatingSystem};
  }
}
```

```dart
import 'dart:convert';
import 'package:shared_preferences/shared_preferences.dart';
import 'package:flutter_secure_storage/flutter_secure_storage.dart';
import '../models/license_info.dart';
import '../utils/constants.dart';

/// 本地存储服务
class StorageService {
  static final FlutterSecureStorage _secureStorage =
      const FlutterSecureStorage();

  /// 保存许可证信息
  static Future<void> saveLicenseInfo(LicenseInfo info) async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.setString(
      AppConstants.licenseDataKey,
      jsonEncode(info.toJson()),
    );
    await prefs.setBool(AppConstants.isActivatedKey, true);

    // Token 保存到安全存储
    if (info.licenseToken != null) {
      await _secureStorage.write(
        key: AppConstants.licenseTokenKey,
        value: info.licenseToken!,
      );
    }
  }

  /// 读取许可证信息
  static Future<LicenseInfo?> getLicenseInfo() async {
    final prefs = await SharedPreferences.getInstance();
    final data = prefs.getString(AppConstants.licenseDataKey);
    if (data == null) return null;

    try {
      final json = jsonDecode(data);
      final token = await _secureStorage.read(key: AppConstants.licenseTokenKey);
      json['license_token'] = token;
      return LicenseInfo.fromJson(json);
    } catch (e) {
      return null;
    }
  }

  /// 检查是否已激活
  static Future<bool> isActivated() async {
    final prefs = await SharedPreferences.getInstance();
    return prefs.getBool(AppConstants.isActivatedKey) ?? false;
  }

  /// 获取许可证Token
  static Future<String?> getLicenseToken() async {
    return await _secureStorage.read(key: AppConstants.licenseTokenKey);
  }

  /// 清除激活信息
  static Future<void> clearLicenseInfo() async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.remove(AppConstants.licenseDataKey);
    await prefs.remove(AppConstants.isActivatedKey);
    await _secureStorage.delete(key: AppConstants.licenseTokenKey);
  }

  /// 保存上次验证时间
  static Future<void> saveLastVerifyTime() async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.setString(
      'last_verify_time',
      DateTime.now().toIso8601String(),
    );
  }

  /// 是否需要重新验证
  static Future<bool> needsReVerification() async {
    final prefs = await SharedPreferences.getInstance();
    final lastTime = prefs.getString('last_verify_time');
    if (lastTime == null) return true;

    final last = DateTime.parse(lastTime);
    final diff = DateTime.now().difference(last).inHours;
    return diff >= AppConstants.verifyIntervalHours;
  }
}
```

```dart
import 'dart:convert';
import 'package:http/http.dart' as http;
import '../models/api_response.dart';
import '../models/license_info.dart';
import '../utils/constants.dart';
import 'device_service.dart';
import 'storage_service.dart';

/// 激活服务 - 处理与后端的通信
class ActivationService {
  static final http.Client _client = http.Client();

  /// 激活
  static Future<ActivationResult> activate({
    required String activationCode,
    String? email,
    String? phone,
  }) async {
    try {
      final deviceId = await DeviceService.getDeviceId();
      final deviceInfo = await DeviceService.getDeviceInfo();

      final response = await _client.post(
        Uri.parse('${AppConstants.baseUrl}${AppConstants.activateEndpoint}'),
        headers: {'Content-Type': 'application/json'},
        body: jsonEncode({
          'activation_code': activationCode,
          'device_id': deviceId,
          'device_info': deviceInfo,
          'email': email,
          'phone': phone,
        }),
      ).timeout(const Duration(seconds: 15));

      final apiResponse = ApiResponse.fromJson(jsonDecode(response.body));

      if (apiResponse.isSuccess && apiResponse.data != null) {
        // 解析许可证信息
        final licenseInfo = LicenseInfo.fromJson(apiResponse.data);

        // 保存到本地
        await StorageService.saveLicenseInfo(licenseInfo);
        await StorageService.saveLastVerifyTime();

        return ActivationResult(
          success: true,
          message: apiResponse.message,
          licenseInfo: licenseInfo,
        );
      } else {
        return ActivationResult(
          success: false,
          message: apiResponse.message,
        );
      }
    } catch (e) {
      return ActivationResult(
        success: false,
        message: '网络连接失败，请检查网络后重试',
      );
    }
  }

  /// 验证许可证(联网验证)
  static Future<ActivationResult> verifyOnline() async {
    try {
      final deviceId = await DeviceService.getDeviceId();

      final response = await _client.post(
        Uri.parse('${AppConstants.baseUrl}${AppConstants.verifyEndpoint}'),
        headers: {'Content-Type': 'application/json'},
        body: jsonEncode({
          'device_id': deviceId,
        }),
      ).timeout(const Duration(seconds: 10));

      final apiResponse = ApiResponse.fromJson(jsonDecode(response.body));

      if (apiResponse.isSuccess && apiResponse.data != null) {
        final isActive = apiResponse.data['is_active'] ?? false;

        if (isActive) {
          final licenseInfo = LicenseInfo.fromJson(apiResponse.data);
          await StorageService.saveLicenseInfo(licenseInfo);
          await StorageService.saveLastVerifyTime();

          return ActivationResult(
            success: true,
            message: '验证成功',
            licenseInfo: licenseInfo,
          );
        }
      }

      // 验证失败,清除本地数据
      await StorageService.clearLicenseInfo();
      return ActivationResult(
        success: false,
        message: '许可证无效或已过期',
      );
    } catch (e) {
      // 网络异常时使用本地验证
      return await verifyOffline();
    }
  }

  /// 离线验证(读取本地缓存)
  static Future<ActivationResult> verifyOffline() async {
    final licenseInfo = await StorageService.getLicenseInfo();

    if (licenseInfo == null) {
      return ActivationResult(
        success: false,
        message: '未找到激活信息',
      );
    }

    // 检查本地过期时间
    if (licenseInfo.isExpired) {
      await StorageService.clearLicenseInfo();
      return ActivationResult(
        success: false,
        message: '许可证已过期',
      );
    }

    return ActivationResult(
      success: true,
      message: '验证成功(离线)',
      licenseInfo: licenseInfo,
    );
  }

  /// 综合验证:优先联网,失败则离线
  static Future<ActivationResult> verify() async {
    final needsOnline = await StorageService.needsReVerification();

    if (needsOnline) {
      return await verifyOnline();
    } else {
      return await verifyOffline();
    }
  }
}

  final LicenseInfo? licenseInfo;

    required this.message,
    this.licenseInfo,
  });
}
```

```dart
import 'package:flutter/foundation.dart';
import '../models/license_info.dart';
import '../services/activation_service.dart';
import '../services/storage_service.dart';

/// 激活状态管理
class ActivationProvider extends ChangeNotifier {
  bool _isLoading = false;
  bool _isActivated = false;
  String? _errorMessage;
  LicenseInfo? _licenseInfo;

  bool get isLoading => _isLoading;
  bool get isActivated => _isActivated;
  String? get errorMessage => _errorMessage;
  LicenseInfo? get licenseInfo => _licenseInfo;

  /// 初始化 - 检查激活状态
  Future<void> initialize() async {
    _isLoading = true;
    notifyListeners();

    try {
      final result = await ActivationService.verify();
      _isActivated = result.success;
      _licenseInfo = result.licenseInfo;
      _errorMessage = result.success ? null : result.message;
    } catch (e) {
      _isActivated = false;
      _errorMessage = '验证失败';
    }

    _isLoading = false;
    notifyListeners();
  }

  /// 激活
  Future<bool> activate(String code, {String? email, String? phone}) async {
    _isLoading = true;
    _errorMessage = null;
    notifyListeners();

    try {
      final result = await ActivationService.activate(
        activationCode: code,
        email: email,
        phone: phone,
      );

      _isActivated = result.success;
      _licenseInfo = result.licenseInfo;
      _errorMessage = result.success ? null : result.message;

      _isLoading = false;
      notifyListeners();
      return result.success;
    } catch (e) {
      _isLoading = false;
      _errorMessage = '激活失败，请稍后重试';
      notifyListeners();
      return false;
    }
  }

  /// 刷新许可证状态
  Future<void> refreshLicense() async {
    final result = await ActivationService.verifyOnline();
    _isActivated = result.success;
    _licenseInfo = result.licenseInfo;
    notifyListeners();
  }

  /// 退出激活(注销)
  Future<void> deactivate() async {
```
